1//! Mechanisms for configuring and managing sharded gateway connections.
2//!
3//! Sharding is a method for load-balancing bots across separate threads or processes. Sharding is
4//! enforced on bots by Discord once they reach a certain number of guilds (2500). Once this
5//! threshold is reached, a but must be sharded such that at most 2500 guilds are allocated per
6//! shard.
7//!
8//! The "recommended" number of guilds per shard is _around_ 1000. Sharding allows for bots to be
9//! distributed by handing shards off to separate processes or even separate machines in a
10//! distributed network (e.g. cloud workers). However, sometimes you may wish for all shards to
11//! share some global state. Serenity accomodates both of these usecases.
12//!
13//! See [Discord's documentation][docs] for more information.
14//!
15//! This module also provides some lower-level facilities for performing sharding manually:
16//!
17//! ### [`ShardManager`]
18//!
19//! The shard manager provides an interface for managing the starting, restarting, and shutdown of
20//! shards by spawning tasks, each containing a [`ShardRunner`], and communicating with each task
21//! using message passing.
22//!
23//! ### [`ShardRunner`]
24//!
25//! The shard runner is responsible for directly running a single shard and communicating with the
26//! gateway through its respective WebSocket client. It performs actions such as identifying,
27//! reconnecting, resuming, and sending presence updates to the gateway.
28//!
29//! [docs]: https://discordapp.com/developers/docs/topics/gateway#sharding
30

67/// An abstract handler for a websocket connection to Discord's gateway.
68///
69/// Allows a user to send and receive messages over said websocket, including:
70///   * setting the current activity
71///   * setting the current online status
72///   * receiving gateway events
73///   * connection management via heartbeating
74///
75/// Shard management (including starting, restarting, heartbeating), is performed by the [`Client`]
76/// automatically on the user's behalf.
77///
78/// # Stand-alone shards
79///
80/// You may instantiate a shard yourself - decoupled from the [`Client`] - by calling
81/// [`Shard::new`]. Most use cases will not necessitate this, and unless you're doing something
82/// really weird you can just let the client do it for you.
83///
84/// **Note**: You _really_ do not need to do this, especially if using multiple shards. Just call
85/// one of the appropriate methods on the [`Client`].
86///
87/// # Examples
88///
89/// See the documentation for [`Self::new`] on how to use this.
90///
91/// [`Client`]: crate::Client
92pub struct Shard {
93    pub client: WsClient,
94    presence: PresenceData,
95    last_heartbeat_sent: Option<Instant>,
96    last_heartbeat_ack: Option<Instant>,
97    heartbeat_interval: Option<std::time::Duration>,
98    application_id_callback: Option<Box<dyn FnOnce(ApplicationId) + Send + Sync>>,
99    /// This is used by the heartbeater to determine whether the last heartbeat was sent without an
100    /// acknowledgement, and whether to reconnect.
101    // This must be set to `true` in `Shard::handle_event`'s `Ok(GatewayEvent::HeartbeatAck)` arm.
102    last_heartbeat_acknowledged: bool,
103    seq: u64,
104    info: ShardInfo,
105    stage: ConnectionStage,
106    /// Instant of when the shard was started.
107    // This acts as a timeout to determine if the shard has - for some reason - not started within
108    // a decent amount of time.
109    pub started: Instant,
110    token: Token,
111    ws_url: Arc<str>,
112    resume_metadata: Option<ResumeMetadata>,
113    compression: TransportCompression,
114    pub intents: GatewayIntents,
115}

122    /// # Examples
123    ///
124    /// Instantiating a new Shard manually for a bot with no shards, and then listening for events:
125    ///
126    /// ```rust,no_run
127    /// use std::num::NonZeroU16;
128    /// use std::sync::Arc;
129    ///
130    /// use serenity::gateway::{Shard, TransportCompression};
131    /// use serenity::model::gateway::{GatewayIntents, ShardInfo};
132    /// use serenity::model::id::ShardId;
133    /// use serenity::secrets::Token;
134    /// use tokio::sync::Mutex;
135    /// #
136    /// # use serenity::http::Http;
137    /// #
138    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
139    /// # let http: Arc<Http> = unimplemented!();
140    /// let token = Token::from_env("DISCORD_TOKEN")?;
141    /// let shard_info = ShardInfo {
142    ///     id: ShardId(0),
143    ///     total: NonZeroU16::MIN,
144    /// };
145    ///
146    /// // retrieve the gateway response, which contains the URL to connect to
147    /// let gateway = Arc::from(http.get_gateway().await?.url);
148    /// let shard = Shard::new(
149    ///     gateway,
150    ///     token,
151    ///     shard_info,
152    ///     GatewayIntents::all(),
153    ///     None,
154    ///     TransportCompression::None,
155    /// )
156    /// .await?;
157    ///
158    /// // at this point, you can create a `loop`, and receive events and match
159    /// // their variants
160    /// # Ok(())
161    /// # }
162    /// ```
163    ///
164    /// # Errors
165    ///
166    /// On Error, will return either [`Error::Gateway`], [`Error::Tungstenite`] or a Rustls/native
167    /// TLS error.
168    pub async fn new(
169        ws_url: Arc<str>,
170        token: Token,
171        info: ShardInfo,
172        intents: GatewayIntents,
173        presence: Option<PresenceData>,
174        compression: TransportCompression,
175    ) -> Result<Shard> {
176        let client = connect(&ws_url, compression).await?;
177
178        let presence = presence.unwrap_or_default();
179        let last_heartbeat_sent = None;
180        let last_heartbeat_ack = None;
181        let heartbeat_interval = None;
182        let last_heartbeat_acknowledged = true;
183        let seq = 0;
184        let stage = ConnectionStage::Handshake;
185
186        Ok(Shard {
187            client,
188            presence,
189            last_heartbeat_sent,
190            last_heartbeat_ack,
191            heartbeat_interval,
192            application_id_callback: None,
193            last_heartbeat_acknowledged,
194            seq,
195            stage,
196            started: Instant::now(),
197            token,
198            info,
199            ws_url,
200            resume_metadata: None,
201            compression,
202            intents,
203        })
204    }

231    /// Sends a heartbeat to the gateway with the current sequence.
232    ///
233    /// This sets the last heartbeat time to now, and [`Self::last_heartbeat_acknowledged`] to
234    /// `false`.
235    ///
236    /// # Errors
237    ///
238    /// Returns [`GatewayError::HeartbeatFailed`] if there was an error sending a heartbeat.
239    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
240    pub async fn heartbeat(&mut self) -> Result<()> {
241        match self.client.send_heartbeat(&self.info, Some(self.seq)).await {
242            Ok(()) => {
243                self.last_heartbeat_sent = Some(Instant::now());
244                self.last_heartbeat_acknowledged = false;
245
246                Ok(())
247            },
248            Err(why) => {
249                if let Error::Tungstenite(err) = &why
250                    && let TungsteniteError::Io(err) = &**err
251                    && err.raw_os_error() != Some(32)
252                {
253                    debug!("[{:?}] Err heartbeating: {:?}", self.info, err);
254                    return Err(Error::Gateway(GatewayError::HeartbeatFailed));
255                }
256
257                warn!("[{:?}] Other err w/ keepalive: {:?}", self.info, why);
258                Err(Error::Gateway(GatewayError::HeartbeatFailed))
259            },
260        }
261    }

307    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
308    fn handle_gateway_dispatch(
309        &mut self,
310        seq: u64,
311        event: DeserializedEvent,
312    ) -> Option<Box<Event>> {
313        if seq > self.seq + 1 {
314            warn!("[{:?}] Sequence off; them: {}, us: {}", self.info, seq, self.seq);
315        }
316
317        self.seq = seq;
318
319        let event = match event {
320            DeserializedEvent::Success(event) => event,
321            DeserializedEvent::Unknown(UnknownEvent {
322                ty,
323                ref data,
324            }) => {
325                debug!("Unknown event: {ty}");
326                debug!("Failing event data: {data:?}");
327                return None;
328            },
329        };
330
331        match &*event {
332            Event::Ready(ready) => {
333                debug!("[{:?}] Received Ready", self.info);
334
335                self.resume_metadata = Some(ResumeMetadata {
336                    session_id: ready.ready.session_id.clone(),
337                    resume_ws_url: ready.ready.resume_gateway_url.clone(),
338                });
339                self.stage = ConnectionStage::Connected;
340
341                if let Some(callback) = self.application_id_callback.take() {
342                    callback(ready.ready.application.id);
343                }
344            },
345            Event::Resumed(_) => {
346                info!("[{:?}] Resumed", self.info);
347
348                self.stage = ConnectionStage::Connected;
349                self.last_heartbeat_acknowledged = true;
350                self.last_heartbeat_sent = Some(Instant::now());
351                self.last_heartbeat_ack = None;
352            },
353            _ => {},
354        }
355
356        Some(event)
357    }

359    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
360    fn handle_gateway_closed(&mut self, data: Option<&CloseFrame>) -> Result<()> {
361        if let Some(code) = data.map(|d| d.code) {
362            match CloseCode(code.into()) {
363                CloseCode::UnknownError => warn!("[{:?}] Unknown gateway error.", self.info),
364                CloseCode::UnknownOpcode => warn!("[{:?}] Sent invalid opcode.", self.info),
365                CloseCode::DecodeError => warn!("[{:?}] Sent invalid message.", self.info),
366                CloseCode::NotAuthenticated => {
367                    warn!("[{:?}] Sent no authentication, or session invalidated.", self.info);
368                    return Err(Error::Gateway(GatewayError::NoAuthentication));
369                },
370                CloseCode::AuthenticationFailed => {
371                    error!(
372                        "[{:?}] Sent invalid authentication, please check the token.",
373                        self.info
374                    );
375
376                    return Err(Error::Gateway(GatewayError::InvalidAuthentication));
377                },
378                CloseCode::AlreadyAuthenticated => {
379                    warn!("[{:?}] Already authenticated.", self.info);
380                },
381                CloseCode::InvalidSequence => {
382                    warn!("[{:?}] Sent invalid seq: {}.", self.info, self.seq);
383                    self.seq = 0;
384                },
385                CloseCode::RateLimited => warn!("[{:?}] Gateway ratelimited.", self.info),
386                CloseCode::SessionTimeout => {
387                    info!("[{:?}] Invalid session.", self.info);
388                    self.resume_metadata = None;
389                },
390                CloseCode::InvalidShard => {
391                    warn!("[{:?}] Sent invalid shard data.", self.info);
392                    return Err(Error::Gateway(GatewayError::InvalidShardData));
393                },
394                CloseCode::ShardingRequired => {
395                    error!("[{:?}] Shard has too many guilds.", self.info);
396                    return Err(Error::Gateway(GatewayError::OverloadedShard));
397                },
398                CloseCode::InvalidApiVersion => {
399                    error!("[{:?}] Invalid gateway API version provided.", self.info);
400                    return Err(Error::Gateway(GatewayError::InvalidApiVersion));
401                },
402                CloseCode::InvalidGatewayIntents => {
403                    error!("[{:?}] Invalid gateway intents have been provided.", self.info);
404                    return Err(Error::Gateway(GatewayError::InvalidGatewayIntents));
405                },
406                CloseCode::DisallowedGatewayIntents => {
407                    error!("[{:?}] Disallowed gateway intents have been provided.", self.info);
408                    return Err(Error::Gateway(GatewayError::DisallowedGatewayIntents));
409                },
410                _ => warn!(
411                    "[{:?}] Unknown close code {}: {:?}",
412                    self.info,
413                    code,
414                    data.map(|d| &d.reason)
415                ),
416            }
417        }
418        Ok(())
419    }

421    /// Handles an event from the gateway over the receiver, requiring the receiver to be passed if
422    /// a reconnect needs to occur.
423    ///
424    /// The best case scenario is that one of two values is returned:
425    /// - `Ok(None)`: a heartbeat, late hello, or session invalidation was received;
426    /// - `Ok(Some((event, None)))`: an op0 dispatch was received, and the shard's voice state will
427    ///   be updated, _if_ the `voice` feature is enabled.
428    ///
429    /// # Errors
430    ///
431    /// Returns a [`GatewayError::InvalidAuthentication`] if invalid authentication was sent in the
432    /// IDENTIFY.
433    ///
434    /// Returns a [`GatewayError::InvalidShardData`] if invalid shard data was sent in the
435    /// IDENTIFY.
436    ///
437    /// Returns a [`GatewayError::NoAuthentication`] if no authentication was sent in the IDENTIFY.
438    ///
439    /// Returns a [`GatewayError::OverloadedShard`] if the shard would have too many guilds
440    /// assigned to it.
441    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
442    pub fn handle_event(&mut self, event: Result<GatewayEvent>) -> Result<Option<ShardAction>> {
443        match event {
444            Ok(GatewayEvent::Dispatch {
445                seq,
446                event,
447            }) => Ok(self.handle_gateway_dispatch(seq, event).map(ShardAction::Dispatch)),
448            Ok(GatewayEvent::Heartbeat) => {
449                info!("[{:?}] Received shard heartbeat", self.info);
450
451                Ok(Some(ShardAction::Heartbeat))
452            },
453            Ok(GatewayEvent::HeartbeatAck) => {
454                self.last_heartbeat_ack = Some(Instant::now());
455                self.last_heartbeat_acknowledged = true;
456
457                trace!("[{:?}] Received heartbeat ack", self.info);
458
459                Ok(None)
460            },
461            Ok(GatewayEvent::Hello(interval)) => {
462                debug!("[{:?}] Received a Hello; interval: {}", self.info, interval);
463
464                if self.stage == ConnectionStage::Resuming {
465                    Ok(None)
466                } else {
467                    self.heartbeat_interval = Some(std::time::Duration::from_millis(interval));
468                    let action = if self.stage == ConnectionStage::Handshake {
469                        ShardAction::Identify
470                    } else {
471                        debug!("[{:?}] Received late Hello; autoreconnecting", self.info);
472                        ShardAction::Reconnect
473                    };
474
475                    Ok(Some(action))
476                }
477            },
478            Ok(GatewayEvent::InvalidateSession(resumable)) => {
479                info!("[{:?}] Received session invalidation", self.info);
480                if !resumable {
481                    self.resume_metadata = None;
482                }
483
484                Ok(Some(ShardAction::Reconnect))
485            },
486            Ok(GatewayEvent::Reconnect) => Ok(Some(ShardAction::Reconnect)),
487            Err(Error::Gateway(GatewayError::Closed(data))) => {
488                self.handle_gateway_closed(data.as_ref())?;
489                Ok(Some(ShardAction::Reconnect))
490            },
491            Err(Error::Tungstenite(why)) => {
492                info!("[{:?}] Websocket error: {:?}", self.info, why);
493                info!("[{:?}] Will attempt to auto-reconnect", self.info);
494
495                Ok(Some(ShardAction::Reconnect))
496            },
497            Err(why) => {
498                warn!("[{:?}] Unhandled error: {:?}", self.info, why);
499                Ok(None)
500            },
501        }
502    }

504    /// Does a heartbeat if needed. Returns false if something went wrong and the shard should be
505    /// restarted.
506    ///
507    /// `true` is returned under one of the following conditions:
508    /// - the heartbeat interval has not elapsed
509    /// - a heartbeat was successfully sent
510    /// - there is no known heartbeat interval yet
511    ///
512    /// `false` is returned under one of the following conditions:
513    /// - a heartbeat acknowledgement was not received in time
514    /// - an error occurred while heartbeating
515    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
516    pub async fn do_heartbeat(&mut self) -> bool {
517        let Some(heartbeat_interval) = self.heartbeat_interval else {
518            // No Hello received yet
519            return self.started.elapsed() < StdDuration::from_secs(15);
520        };
521
522        // If a duration of time less than the heartbeat_interval has passed, then don't perform a
523        // keepalive or attempt to reconnect.
524        if let Some(last_sent) = self.last_heartbeat_sent
525            && last_sent.elapsed() <= heartbeat_interval
526        {
527            return true;
528        }
529
530        // If the last heartbeat didn't receive an acknowledgement, then auto-reconnect.
531        if !self.last_heartbeat_acknowledged {
532            debug!("[{:?}] Last heartbeat not acknowledged", self.info,);
533
534            return false;
535        }
536
537        // Otherwise, we're good to heartbeat.
538        if let Err(why) = self.heartbeat().await {
539            warn!("[{:?}] Err heartbeating: {:?}", self.info, why);
540
541            false
542        } else {
543            trace!("[{:?}] Heartbeat", self.info);
544
545            true
546        }
547    }

549    /// Calculates the heartbeat latency between the shard and the gateway.
550    // Shamelessly stolen from brayzure's commit in eris:
551    // <https://github.com/abalabahaha/eris/commit/0ce296ae9a542bcec0edf1c999ee2d9986bed5a6>
552    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
553    pub fn latency(&self) -> Option<StdDuration> {
554        if let Some(sent) = self.last_heartbeat_sent
555            && let Some(received) = self.last_heartbeat_ack
556            && received > sent
557        {
558            return Some(received - sent);
559        }
560
561        None
562    }

629    /// Reinitializes an existing WebSocket client, replacing it.
630    ///
631    /// This will set the stage of the shard before and after instantiation of the client.
632    ///
633    /// # Errors
634    ///
635    /// Errors if unable to establish a websocket connection.
636    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
637    pub async fn reinitialize(&mut self) -> Result<WsClient> {
638        debug!("[{:?}] Initializing.", self.info);
639
640        // Reconnect to the resume URL if possible, otherwise use the generic URL.
641        let ws_url = self
642            .resume_metadata
643            .as_ref()
644            .map_or(self.ws_url.as_ref(), |m| m.resume_ws_url.as_ref());
645
646        // We need to do two, sort of three things here:
647        // - set the stage of the shard as opening the websocket connection
648        // - open the websocket connection
649        // - if successful, set the current stage as Handshaking
650        //
651        // This is used to accurately assess whether the state of the shard is accurate when a
652        // Hello is received.
653        self.stage = ConnectionStage::Connecting;
654        self.started = Instant::now();
655        let client = connect(ws_url, self.compression).await?;
656        self.stage = ConnectionStage::Handshake;
657
658        Ok(client)
659    }
660
661    /// # Errors
662    ///
663    /// Errors if unable to re-establish a websocket connection.
664    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
665    pub async fn resume(&mut self) -> Result<()> {
666        debug!("[{:?}] Attempting to resume", self.info);
667
668        self.client = self.reinitialize().await?;
669        self.stage = ConnectionStage::Resuming;
670
671        if let Some(m) = &self.resume_metadata {
672            self.client
673                .send_resume(&self.info, &m.session_id, self.seq, self.token.expose_secret())
674                .await
675        } else {
676            Err(Error::Gateway(GatewayError::NoSessionId))
677        }
678    }
679
680    /// # Errors
681    ///
682    /// Errors if there is a problem with the WS connection.
683    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
684    pub async fn update_presence(&mut self) -> Result<()> {
685        self.client.send_presence_update(&self.info, &self.presence).await
686    }
687}
688
689async fn connect(base_url: &str, compression: TransportCompression) -> Result<WsClient> {
690    let url = Url::parse(&aformat!(
691        "{}?v={}{}",
692        CapStr::<64>(base_url),
693        constants::GATEWAY_VERSION,
694        compression.query_param()
695    ))
696    .map_err(|why| {
697        warn!("Error building gateway URL with base `{base_url}`: {why:?}");
698        Error::Gateway(GatewayError::BuildingUrl)
699    })?;
700
701    WsClient::connect(url, compression).await
702}

746/// Indicates the current connection stage of a [`Shard`].
747///
748/// This can be useful for knowing which shards are currently "down"/"up".
749#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd, Serialize)]
750#[non_exhaustive]
751pub enum ConnectionStage {
752    /// Indicator that the [`Shard`] is normally connected and is not in, e.g., a resume phase.
753    Connected,
754    /// Indicator that the [`Shard`] is connecting and is in, e.g., a resume phase.
755    Connecting,
756    /// Indicator that the [`Shard`] is fully disconnected and is not in a reconnecting phase.
757    Disconnected,
758    /// Indicator that the [`Shard`] is currently initiating a handshake.
759    Handshake,
760    /// Indicator that the [`Shard`] has sent an IDENTIFY packet and is awaiting a READY packet.
761    Identifying,
762    /// Indicator that the [`Shard`] has sent a RESUME packet and is awaiting a RESUMED packet.
763    Resuming,
764}
765
766impl ConnectionStage {
767    /// Whether the stage is a form of connecting.
768    ///
769    /// This will return `true` on:
770    /// - [`Connecting`][`ConnectionStage::Connecting`]
771    /// - [`Handshake`][`ConnectionStage::Handshake`]
772    /// - [`Identifying`][`ConnectionStage::Identifying`]
773    /// - [`Resuming`][`ConnectionStage::Resuming`]
774    ///
775    /// All other variants will return `false`.
776    ///
777    /// # Examples
778    ///
779    /// Assert that [`ConnectionStage::Identifying`] is a connecting stage:
780    ///
781    /// ```rust
782    /// use serenity::gateway::ConnectionStage;
783    ///
784    /// assert!(ConnectionStage::Identifying.is_connecting());
785    /// ```
786    ///
787    /// Assert that [`ConnectionStage::Connected`] is _not_ a connecting stage:
788    ///
789    /// ```rust
790    /// use serenity::gateway::ConnectionStage;
791    ///
792    /// assert!(!ConnectionStage::Connected.is_connecting());
793    /// ```
794    #[must_use]
795    pub fn is_connecting(self) -> bool {
796        use self::ConnectionStage::{Connecting, Handshake, Identifying, Resuming};
797        matches!(self, Connecting | Handshake | Identifying | Resuming)
798    }
799}

801impl fmt::Display for ConnectionStage {
802    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
803        f.write_str(match *self {
804            Self::Connected => "connected",
805            Self::Connecting => "connecting",
806            Self::Disconnected => "disconnected",
807            Self::Handshake => "handshaking",
808            Self::Identifying => "identifying",
809            Self::Resuming => "resuming",
810        })
811    }
812}

828/// The transport compression method to use.
829#[derive(Copy, Clone, Debug, PartialEq, Eq)]
830#[non_exhaustive]
831pub enum TransportCompression {
832    /// No transport compression. Payload compression will be used instead.
833    None,
834
835    #[cfg(feature = "transport_compression_zlib")]
836    /// Use zlib-stream transport compression.
837    Zlib,
838
839    #[cfg(feature = "transport_compression_zstd")]
840    /// Use zstd-stream transport compression.
841    Zstd,
842}
843
844impl TransportCompression {
845    fn query_param(self) -> ArrayString<21> {
846        #[cfg_attr(
847            not(any(
848                feature = "transport_compression_zlib",
849                feature = "transport_compression_zstd"
850            )),
851            expect(unused_mut)
852        )]
853        let mut res = ArrayString::new();
854        match self {
855            Self::None => {},
856            #[cfg(feature = "transport_compression_zlib")]
857            Self::Zlib => aformat_into!(res, "&compress=zlib-stream"),
858            #[cfg(feature = "transport_compression_zstd")]
859            Self::Zstd => aformat_into!(res, "&compress=zstd-stream"),
860        }
861
862        res
863    }
864}
