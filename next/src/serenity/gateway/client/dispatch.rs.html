40/// Calls the user's event handlers and the framework handler.
41pub(crate) async fn dispatch_model(
42    event: Box<Event>,
43    context: Context,
44    #[cfg(feature = "framework")] framework: Option<Arc<dyn Framework>>,
45    event_handler: Option<Arc<dyn EventHandler>>,
46    raw_event_handler: Option<Arc<dyn RawEventHandler>>,
47) {
48    if let Some(raw_handler) = raw_event_handler {
49        raw_handler.raw_event(context.clone(), &event).await;
50    }
51
52    let (full_event, extra_event) = update_cache_with_event(
53        #[cfg(feature = "cache")]
54        &context.cache,
55        event,
56    );
57
58    spawn_named("dispatch::user", async move {
59        #[cfg(feature = "framework")]
60        tokio::join!(
61            dispatch_framework(&context, framework, &full_event, extra_event.as_ref()),
62            dispatch_event_handler(&context, event_handler, &full_event, extra_event.as_ref())
63        );
64
65        #[cfg(not(feature = "framework"))]
66        dispatch_event_handler(&context, event_handler, &full_event, extra_event.as_ref()).await;
67    });
68}
69
70#[cfg(feature = "framework")]
71async fn dispatch_framework(
72    context: &Context,
73    framework: Option<Arc<dyn Framework>>,
74    full_event: &FullEvent,
75    extra_event: Option<&FullEvent>,
76) {
77    if let Some(framework) = framework {
78        if let Some(extra_event) = extra_event {
79            framework.dispatch(context, extra_event).await;
80        }
81
82        framework.dispatch(context, full_event).await;
83    }
84}

86async fn dispatch_event_handler(
87    context: &Context,
88    event_handler: Option<Arc<dyn EventHandler>>,
89    full_event: &FullEvent,
90    extra_event: Option<&FullEvent>,
91) {
92    if let Some(handler) = event_handler {
93        if let Some(extra_event) = extra_event {
94            handler.dispatch(context, extra_event).await;
95        }
96
97        handler.dispatch(context, full_event).await;
98    }
99}

101/// Updates the cache with the incoming event data and builds the full event data out of it.
102///
103/// Can return a secondary [`FullEvent`] for "virtual" events like [`FullEvent::CacheReady`] or
104/// [`FullEvent::ShardsReady`]. Secondary events are traditionally dispatched first.
105///
106/// Can return `None` if an event is unknown.

125        },
126        Event::AutoModActionExecution(event) => FullEvent::AutoModActionExecution {
127            execution: event.execution,
128        },
129        Event::ChannelCreate(mut event) => {
130            update_cache!(cache, event);
131
132            let channel = event.channel;
133            if channel.base.kind == ChannelType::Category {
134                FullEvent::CategoryCreate {
135                    category: channel,
136                }
137            } else {
138                FullEvent::ChannelCreate {
139                    channel,
140                }
141            }
142        },
143        Event::ChannelDelete(mut event) => {
